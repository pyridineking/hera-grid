cmake_minimum_required(VERSION 3.19)
project(HeraGrid LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(NEWGLOBALFITTING_DIR "/Data4/pyridineking/GlobalFitting/newglobalfitting" CACHE PATH "Original newglobalfitting source directory")
find_package(Threads REQUIRED)
find_package(yaml-cpp REQUIRED)
find_program(LHAPDF_CONFIG lhapdf-config REQUIRED)
execute_process(COMMAND ${LHAPDF_CONFIG} --prefix OUTPUT_VARIABLE LHAPDF_PREFIX OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
find_path(LHAPDF_INCLUDE LHAPDF/LHAPDF.h HINTS ${LHAPDF_PREFIX}/include REQUIRED)
find_library(LHAPDF_LIBRARY LHAPDF HINTS ${LHAPDF_PREFIX}/lib REQUIRED)

# Only build-local copies are modified. The external fitting project is read-only.
set(CTEQ_ORIGINAL "${NEWGLOBALFITTING_DIR}/CTEQDisPac")
set(CTEQ_LOCAL "${CMAKE_CURRENT_BINARY_DIR}/cteq")
file(MAKE_DIRECTORY "${CTEQ_LOCAL}")
set(CTEQ_FILES CTEQDisPac.cxx SfZm05.cxx Acot05.cxx GotSK1Q.cxx ZmNNLO.cxx GmJS.cxx GmJSa.h CTEQDisPac.h)
set(PROVENANCE "")
foreach(name IN LISTS CTEQ_FILES)
  configure_file("${CTEQ_ORIGINAL}/${name}" "${CTEQ_LOCAL}/${name}" COPYONLY)
  file(SHA256 "${CTEQ_ORIGINAL}/${name}" digest)
  string(APPEND PROVENANCE "${name} ${digest}\n")
endforeach()
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/backend_sources.sha256" "${PROVENANCE}")
string(SHA256 BACKEND_DIGEST "${PROVENANCE}")
file(READ "${CTEQ_LOCAL}/CTEQDisPac.h" header)
set(anchor "    int GetOrder() const")
string(FIND "${header}" "${anchor}" position)
if(position EQUAL -1)
  message(FATAL_ERROR "CTEQDisPac interface changed: review the local integration-settings adapter")
endif()
string(REPLACE "${anchor}" "    void SetGridIntegration(double accuracy, unsigned depth) { m_int_accuracy = accuracy; m_max_depth = depth; }\n\n${anchor}" header "${header}")
file(WRITE "${CTEQ_LOCAL}/CTEQDisPac.h" "${header}")
# Upstream initializes iBsn[2][5] in a loop running to 7: rows 5/6 are assembled below.
# Restrict that primitive-boson loop to its five valid entries in the local copy.
file(READ "${CTEQ_LOCAL}/CTEQDisPac.cxx" cteq_source)
string(REPLACE "iint < m_NInteractions" "iint < 5" cteq_source "${cteq_source}")
file(WRITE "${CTEQ_LOCAL}/CTEQDisPac.cxx" "${cteq_source}")
add_library(hera_cteq STATIC ${CTEQ_LOCAL}/CTEQDisPac.cxx ${CTEQ_LOCAL}/SfZm05.cxx ${CTEQ_LOCAL}/Acot05.cxx ${CTEQ_LOCAL}/GotSK1Q.cxx ${CTEQ_LOCAL}/ZmNNLO.cxx ${CTEQ_LOCAL}/GmJS.cxx)
target_include_directories(hera_cteq SYSTEM PUBLIC "${CTEQ_LOCAL}")
add_library(hera_grid STATIC src/grid.cpp)
set_target_properties(hera_grid PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(hera_grid PUBLIC include)
target_compile_options(hera_grid PRIVATE -Wall -Wextra -Wpedantic)
add_library(hera_grid_builder STATIC src/builder.cpp src/config.cpp)
target_link_libraries(hera_grid_builder PUBLIC hera_grid hera_cteq yaml-cpp Threads::Threads)
target_compile_definitions(hera_grid_builder PRIVATE HERA_BACKEND_DIGEST="${BACKEND_DIGEST}")
target_compile_options(hera_grid_builder PRIVATE -Wall -Wextra -Wpedantic)
add_executable(hera-grid src/main.cpp)
target_include_directories(hera-grid PRIVATE ${LHAPDF_INCLUDE})
target_link_libraries(hera-grid PRIVATE hera_grid_builder ${LHAPDF_LIBRARY})
target_compile_options(hera-grid PRIVATE -Wall -Wextra -Wpedantic)
